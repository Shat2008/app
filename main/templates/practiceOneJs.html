Міні-проєкт: калькулятор

У цьому проєкті ми створимо простий калькулятор для додавання, віднімання, множення та ділення чисел.

� Як це працює - Приклад коду

HTML структура:

<div class="calculator">
    <input type="number" id="num1" placeholder="Перше число">
    <select id="operation">
        <option value="add">+</option>
        <option value="subtract">-</option>
        <option value="multiply">×</option>
        <option value="divide">÷</option>
    </select>
    <input type="number" id="num2" placeholder="Друге число">
    <button onclick="calculate()">Обчислити</button>
    <p id="result">Результат: —</p>
</div>

JavaScript функція:

function calculate() {
    const n1 = Number(document.getElementById('num1').value);
    const n2 = Number(document.getElementById('num2').value);
    const op = document.getElementById('operation').value;
    let res;

    if(op === "add") res = n1 + n2;
    else if(op === "subtract") res = n1 - n2;
    else if(op === "multiply") res = n1 * n2;
    else if(op === "divide") {
        res = n2 !== 0 ? n1 / n2 : "Помилка: ділення на 0";
    }

    document.getElementById('result').innerText = "Результат: " + res;
}
